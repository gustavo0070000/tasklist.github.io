                    b.classList.remove('active-owner', b.dataset.activeBg, b.dataset.activeText, b.dataset.activeBorder);
                    b.classList.add('border-outline-variant', 'text-on-surface-variant');

                this.classList.add('active-owner', this.dataset.activeBg, this.dataset.activeText, this.dataset.activeBorder);
                this.classList.remove('border-outline-variant', 'text-on-surface-variant');
            });
        });

        priorityBtns.forEach(btn => {

                priorityBtns.forEach(b => {
                    b.classList.remove(b.dataset.activeBg, b.dataset.activeBorder);
                    b.classList.add('border-outline-variant');
                });

            });
